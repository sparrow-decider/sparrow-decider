cmake_minimum_required(VERSION 3.16)
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../../sparrow")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
if(NOT SPARROW_WIDE)  # nano: 128-d -> smaller working memory
  idf_build_set_property(COMPILE_DEFINITIONS "SPARROW_MAX_DIM=128" APPEND)
  if(NOT SPARROW_HIDDEN)
    set(SPARROW_HIDDEN 256)
  endif()
  idf_build_set_property(COMPILE_DEFINITIONS "SPARROW_MAX_HIDDEN=${SPARROW_HIDDEN}" APPEND)
endif()
if(SPARROW_MLP_INTERNAL)
  idf_build_set_property(COMPILE_DEFINITIONS "SPARROW_MLP_INTERNAL=1" APPEND)
endif()
if(SPARROW_BLOB_PSRAM)
  idf_build_set_property(COMPILE_DEFINITIONS "SPARROW_BLOB_PSRAM=1" APPEND)
endif()
if(SPARROW_PROFILE)
  idf_build_set_property(COMPILE_DEFINITIONS "SPARROW_PROFILE=1" APPEND)
endif()
if(SPARROW_INT8)
  idf_build_set_property(COMPILE_DEFINITIONS "SPARROW_INT8_ACT=1" APPEND)
endif()
project(sparrow_bench)
